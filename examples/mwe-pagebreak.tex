% Minimal example: page breaks.
% Cases 1 and 2 start from the same page and differ only by the height of
% the image (7 cm, then 8 cm).
%  1. The image just fits at the bottom of the page: it stays there with the
%     lines beside it, and the rest of the block runs full width on the next
%     page.
%  2. The image does not fit: the block moves to the next page, together with
%     the heading before it.
%  3. A long block: the text below the image runs on over several pages.
\documentclass{article}
\usepackage{graphicx}
\usepackage{kantlipsum}
\usepackage{wrapblock}
\setlength{\emergencystretch}{1em} % the lines beside the images are narrow
\begin{document}

\kant[1-2]

\section{Case 1: the image fits}

\begin{wrapblock}{\includegraphics[height=7cm]{example-image-9x16}}
\kant[4-6]
\end{wrapblock}

\clearpage
\kant[1-2]

\section{Case 2: one centimetre taller}

\begin{wrapblock}{\includegraphics[height=8cm]{example-image-9x16}}
\kant[4-6]
\end{wrapblock}

\clearpage
\section{Case 3: a long block}

\begin{wrapblock}[side=l]{\includegraphics[width=4cm]{example-image-b}}
\kant[9-14]
\end{wrapblock}

\end{document}
