% L2: a tall line after the k-th sentence, sweeping the end of the zone.
\newcommand\sweep[1]{%
  \begin{wrapblock}{\img[2.5cm]}
  \count255=0
  \loop\ifnum\count255<#1 \sentence\advance\count255 1 \repeat
  \tall\ \sentence\sentence\sentence\sentence
  \end{wrapblock}
  \para\par}
\scene{L2 --- tall line at the end of the zone, k = 2 to 4}
\sweep{2}\sweep{3}\sweep{4}
\scene{L2 --- k = 5 to 7}
\sweep{5}\sweep{6}\sweep{7}
