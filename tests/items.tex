% Items of a list in a block, the list being opened before the block and
% closed after it; image on the right.
%  T1. in the middle of a page;
%  T2. near the bottom, long items: the image is deferred;
%  T3. near the bottom, short items: the whole block moves, and starts at
%      the top of the next page, with the indentation of the list.
% run.zsh checks that there is no error and that no word overlaps an image.
\documentclass{article}
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}\usepackage{lmodern}
\fi
\usepackage{graphicx,kantlipsum}
\usepackage{wrapblock}
\newcommand\img[2]{\fbox{\parbox[t][#2][s]{#1}{\tiny IMGA\hfill IMGB\par\vfill IMGC}}}
\newcommand\case[3]{% lines before, image height, items
  \clearpage
  #1
  \begin{enumerate}
  \item An item before the block.
  \begin{wrapblock}{\img{4cm}{#2}}
  #3
  \end{wrapblock}
  \item An item after the block. \kant[2][3]
  \end{enumerate}}
\begin{document}
\case{\kant[1]}{3cm}{\item First item. \kant[1][1-2] \item Second item. \kant[1][3-4] \item Third item. \kant[2]}
\case{\kant[1-3]\par\kant[4][1-1]}{7cm}{\item First item. \kant[1] \item Second item. \kant[2] \item Third item. \kant[3]}
\case{\kant[1-3]\par\kant[4][1-1]}{7cm}{\item A short first item. \item A short second item.}
\end{document}
